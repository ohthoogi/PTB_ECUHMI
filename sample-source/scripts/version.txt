//---------------------------------------------------------------------------------//
//	HMI <-> ECU
//	
//	000 이 숫자 표시기로 표시가 안됨
//	major 에 1000을 곱한 뒤 minor 를 더해  1000 을 소수점 자리 고정 기능을 사용해서
//	minor 를 소수점으로 표시 해야 스크린에서 000을 볼 수가 있다
//	소수점 자리 고정은 3으로 해야 000을 전부 볼수가 있다
//
//	숫자 표시기의 소수점 자리 고정 기능은 소수점 숫자가 있을 때 표시 해주는게 아닌
//	입력된 숫자의 소수점 위치만 옮긴다
//	ex) 	1000 => 1.000
//			V 1.000
//			int major = 1
//			int minor = 000
//			(major * 1000) + minor = 1000
//			표시 => 1.000
//---------------------------------------------------------------------------------//

int major = 8;
int minor = 42;

if(@[X:HW09870.1] == 1 && @[X:HW09870.2] == 1){
	@[X:HW09870.0] = 1;
	@[X:HW09870.1] = 0;
	@[W:HW09875] = 0;
}

@[W:HW09872] = (major*1000)+minor;
