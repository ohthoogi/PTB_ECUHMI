		// transform low byte to 2 ascii data

		HEX2ASCII((fullByte & 0x0f) , &alphabet[1], 1);
		
		// full byte index = low byte index
		// high byte index value = default value = 0x30 0x30
		data2send[14] = alphabet[0];
		data2send[15] = alphabet[1];
	} 
}
// reset button pressed
else if(@[X:HW65100.0] == 1) {
	// data address for reset button
	data2send[11] = 0x35; 
}
// if nothing pressed - default
else {
	@[W:HW62347] = 1;
	data2send[11] = 0x30; 
	
	// send message 2
	if(@[X:HW65243.0] == 1){
		data2send[11] = 0x36;
	}

//----------------------------

if(@[W:HW62345] == 2){

	if(@[X:HW19812.3] != @[X:HS0000.4]){
		// timer counter
		@[W:HW62348]++;
		// set new timer value 
		@[X:HW19812.3] = @[X:HS0000.4];
	}
	// reset after 150ms
	if(@[W:HW62348] > 4){
		// transmit flag
		@[W:HW62345] = 0;
		// send default page message
		@[W:HW62347] = 1;
		// reset timer counter
		@[W:HW62348] = 0;
		// send default page command
		data2send[11] = 0x30; 
		@[X:HW65243.0] = 0;
	}
}

//---------------------------- 
//	transmit mode
//---------------------------- 

// when transmit ready
if(@[W:HW62345] == 0){
	// transmit data
	size232 = WriteToIO(0,&data2send[0], 27);
	
	// if button is pressed
	if(@[W:HW62347] == 0) {
		// initialize timer value
		@[X:HW19812.3] = @[X:HS0000.4];
		// prevent entering receive mode
		@[W:HW62345] = 2;
	} else {

		@[W:HW62345] = 1;
			
		// reset buttons

		if(@[X:HW61100.4] == 1){
			@[X:HW61100.4] = 0;
			// pop up 5 on
			@[X:HW39111.0] = 1;
			// string message 17 - "save complete"
			@[W:HW33333] = 17;
			// change'이전 설정 값' to '현재 설정 값' - 하한
			@[W:HW61110] = @[W:HW61115];
			// change'이전 설정 값' to '현재 설정 값' - 상한
			@[W:HW61111] = @[W:HW61116];
		}	
		// button 2
		if(@[X:HW62100.4] == 1){
			@[X:HW62100.4] = 0;
			// pop up 5 on
			@[X:HW39111.0] = 1;
			// string message 17 - "save complete"
			@[W:HW33333] = 17;
			// change'이전 설정 값' to '현재 설정 값' - 하한
			@[W:HW61120] = @[W:HW61125];
			// change'이전 설정 값' to '현재 설정 값' - 상한
			@[W:HW61121] = @[W:HW61126];
		}
		// button 3
		if(@[X:HW63100.4] == 1){
			@[X:HW63100.4] = 0;
			// pop up 5 on
			@[X:HW39111.0] = 1;
			// string message 17 - "save complete"
			@[W:HW33333] = 17;
			// change'이전 설정 값' to '현재 설정 값' - 하한
			@[W:HW61130] = @[W:HW61135];
			// change'이전 설정 값' to '현재 설정 값' - 상한
			@[W:HW61131] = @[W:HW61136];
		}
		// button 4
		if(@[X:HW64100.4] == 1){
			@[X:HW64100.4] = 0;
			// pop up 5 on
			@[X:HW39111.0] = 1;
			// string message 17 - "save complete"
			@[W:HW33333] = 17;

			@[W:HW61140] = @[W:HW61145];
			// change'이전 설정 값' to '현재 설정 값' - 상한

			@[X:HW64100.5] = 0;

// when first byte is 0x65, which is sof	
if(global_Response[0] == 0x65){
	// when last byte is 0x0d, which is eof
	if(global_Response[26] == 0x0d){
		// receive complete ready to send
		@[W:HW62345] = 0;
		// timeout reset
		@[W:HW62344] = 0;
		
		// message 1/2 converter
		@[X:HW65243.0] = !@[X:HW65243.0];
		
		// connection check reset
		@[W:HW19811] = 0;
		
		// reset button entering flag
		@[W:HW62347] = 0;

		
		// get data only
		for(global_rcv_cnt = 10; global_rcv_cnt < 26; global_rcv_cnt++){
			// if alphabet data
			if( global_Response[global_rcv_cnt] > 0x60){
				global_buffer[global_buffercnt++] = global_Response[global_rcv_cnt] - 0x57;
			}
			// if numeric data   
			else {
				global_buffer[global_buffercnt++] = global_Response[global_rcv_cnt] - 0x30;
			}
		}
		// received address
		global_addr = (global_buffer[0] << 4) + global_buffer[1];
		
		// oil temp page data 1 / page 6 / 하한 값
		////------------------------------ a ------------------------------////
		if(global_addr == 0x5a) {					
			// 오일탱크 온도							
			@[W:HW61110] = (global_buffer[2] << 4) + global_buffer[3];

			@[W:HW61130] = (global_buffer[6] << 4) + global_buffer[7];

			@[W:HW61140] = (global_buffer[8] << 4) + global_buffer[9];

			//	set '현재 설정 값' to '이전 설정 값'
			//---------------------------- 
			if(@[X:HW61100.5] == 0) {
				@[W:HW61115] = @[W:HW61110];
			}
			if(@[X:HW62100.5] == 0) {
				@[W:HW61125] = @[W:HW61120];
			}

				@[W:HW61135] = @[W:HW61130];
			}
			if(@[X:HW64100.5] == 0) {
				@[W:HW61145] = @[W:HW61140];
			}
			//---------------------------- 
		}
		
		// oil temp page data 2 / page 6 / 상한 값
		////------------------------------ b ------------------------------////
		else if(global_addr == 0x5b) {		
			int temp1 = 0;
			int temp2 = 0;				
			// 오일탱크 온도						
			@[W:HW61111] = (global_buffer[2] << 4) + global_buffer[3];
			// 1번 쿨러 온도
			@[W:HW61121] = (global_buffer[4] << 4) + global_buffer[5];
			// 2번 쿨러 온도
			@[W:HW61131] = (global_buffer[6] << 4) + global_buffer[7];
			// 3번 쿨러 온도
			@[W:HW61141] = (global_buffer[8] << 4) + global_buffer[9];
				
			//---------------------------- 
			//	set '현재 설정 값' to '이전 설정 값'

			if(@[X:HW61100.5] == 0) {
				@[W:HW61116] = @[W:HW61111];

		}
	}
