		global_ReadSize = ReadFromIO(0,&global_Response[global_ReadSize], 27-global_ReadSize );
	}

		// send message

		// get high & low byte
		highByte = HIBYTE(@[W:HW41115]);
		lowByte= LOBYTE(@[W:HW41115]);
		
		// transform high byte to 2 ascii data
		HEX2ASCII(((highByte & 0xff) >> 4) , &alphabet[0], 1);
		HEX2ASCII((highByte & 0x0f) , &alphabet[1], 1);
		
		// transform low byte to 2 ascii data
		HEX2ASCII(((lowByte & 0xff) >> 4) , &alphabet[2], 1);
		HEX2ASCII((lowByte & 0x0f) , &alphabet[3], 1);
		
		// high byte index
		data2send[16] = alphabet[0];
		data2send[17] = alphabet[1];
		
		// low byte index
		data2send[18] = alphabet[2];
		data2send[19] = alphabet[3];

		fullByte = @[W:HW41115];

		HEX2ASCII((fullByte & 0x0f) , &alphabet[1], 1);
		
		// full byte index = low byte index
		// high byte index value = default value = 0x30 0x30
		data2send[18] = alphabet[0];
		data2send[19] = alphabet[1];
	}

		HEX2ASCII((highByte & 0x0f) , &alphabet[1], 1);
		
		// transform low byte to 2 ascii data
		HEX2ASCII(((lowByte & 0xff) >> 4) , &alphabet[2], 1);
		HEX2ASCII((lowByte & 0x0f) , &alphabet[3], 1);
		
		// high byte index
		data2send[12] = alphabet[0];
		data2send[13] = alphabet[1];
		
		// low byte index
		data2send[14] = alphabet[2];
		data2send[15] = alphabet[3];	
		
	}

// when first byte is 0x65, which is sof	
if(global_Response[0] == 0x65){
	// when last byte is 0x0d, which is eof
	if(global_Response[26] == 0x0d){
		// receive complete ready to send
		@[W:HW42345] = 0;
		// timeout reset
		@[W:HW42344] = 0;
		
		// message 1/2 converter

		// get data only
		for(global_rcv_cnt = 10; global_rcv_cnt < 26; global_rcv_cnt++){
			// if alphabet data
			if( global_Response[global_rcv_cnt] > 0x60){
				global_buffer[global_buffercnt++] = global_Response[global_rcv_cnt] - 0x57;
			}
			// if numeric data  
			else {
				global_buffer[global_buffercnt++] = global_Response[global_rcv_cnt] - 0x30;
			}
		}
		// received address
		global_addr = (global_buffer[0] << 4) + global_buffer[1];
		/*
		// debug 260921
		@[W:HW01280] = (global_buffer[0] << 4) + global_buffer[1];

		@[W:HW01286] = (global_buffer[12] << 4) + global_buffer[13];
		@[W:HW01287] = (global_buffer[14] << 4) + global_buffer[15];
		*/
		int temp1 = 0;
		int temp2 = 0;	
		// pressure page data 1 / page 4 / 하한 값
		////------------------------------ a ------------------------------////
		if(global_addr == 0x3a) {		
			// 보조 펌프 압력
			@[W:HW41110] = (global_buffer[2] << 4) + global_buffer[3];	
			
			// 주 펌프 압력 저압
			@[W:HW41120] = (global_buffer[4] << 4) + global_buffer[5];
			
			// 주 펌프 압력 고압
			temp1 = (global_buffer[6] << 4) + global_buffer[7];
			temp2 = (global_buffer[8] << 4) + global_buffer[9];	
			@[W:HW41130] = (temp1 << 8) + temp2;

			@[W:HW41140] = (global_buffer[10] << 4) + global_buffer[11];
			
			//브레이크 off 압력
			@[W:HW41150] = (global_buffer[12] << 4) + global_buffer[13];

				@[W:HW41115] = @[W:HW41110];
			}

				@[W:HW41125] = @[W:HW41120];
			}
			if(@[X:HW43100.5] == 0) {
				@[W:HW41135] = @[W:HW41130];
			}
			if(@[X:HW44100.5] == 0) {
				@[W:HW41145] = @[W:HW41140];
			}
			if(@[X:HW45100.5] == 0){
				@[W:HW41155] = @[W:HW41150];
			}
			//---------------------------- 
		} 
		////------------------------------ a ------------------------------////
		
		// pressure page data 2 / page 4 / 상한 값
		////------------------------------ b ------------------------------////
		else if(global_addr == 0x3b) {			
			// 보조 펌프 압력							
			@[W:HW41111] = (global_buffer[2] << 4) + global_buffer[3];	
			
			// 주 펌프 압력 저압
			@[W:HW41121] = (global_buffer[4] << 4) + global_buffer[5];

			temp2 = (global_buffer[8] << 4) + global_buffer[9];	
			@[W:HW41131] = (temp1 << 8) + temp2;
			@[W:HW41185] = 0x08;
			@[W:HW41186] = 0x0c;
			@[W:HW41187] = (0x08 << 4)+0x0c;
